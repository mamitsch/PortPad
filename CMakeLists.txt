cmake_minimum_required(VERSION 3.16)
project(PortPad VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.14)
pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)

add_library(portpad_state src/AppState.cpp src/FileBrowser.cpp src/FileSystem.cpp src/Utf8.cpp src/TextBuffer.cpp src/DocumentIO.cpp src/VirtualKeyboard.cpp src/EditorState.cpp)
target_include_directories(portpad_state PUBLIC include)
add_library(portpad_ui src/Input.cpp src/Keymap.cpp src/Renderer.cpp src/EditorRenderer.cpp)
target_include_directories(portpad_ui PUBLIC include)
target_link_libraries(portpad_ui PUBLIC portpad_state PkgConfig::SDL2 PkgConfig::SDL2_TTF)
add_executable(portpad src/main.cpp)
target_link_libraries(portpad PRIVATE portpad_ui)
file(COPY assets DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
# Seed a runtime config without overwriting user edits on reconfiguration.
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/config/keymap.ini")
    file(COPY config DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(editor_render_tests tests/editor_render_tests.cpp)
    target_link_libraries(editor_render_tests PRIVATE portpad_ui)
    target_compile_definitions(editor_render_tests PRIVATE PORTPAD_TEST_KEYMAP="${CMAKE_CURRENT_SOURCE_DIR}/config/keymap.ini" PORTPAD_TEST_FONT="${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/DejaVuSans.ttf")
    add_test(NAME editor_render COMMAND editor_render_tests)
    set_tests_properties(editor_render PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    add_executable(editor_tests tests/editor_tests.cpp)
    target_link_libraries(editor_tests PRIVATE portpad_state)
    add_test(NAME editor COMMAND editor_tests)
    add_executable(browser_tests tests/browser_tests.cpp)
    target_link_libraries(browser_tests PRIVATE portpad_state)
    add_test(NAME browser COMMAND browser_tests)
    add_executable(state_tests tests/state_tests.cpp)
    target_link_libraries(state_tests PRIVATE portpad_state)
    add_test(NAME state COMMAND state_tests)
    add_executable(input_tests tests/input_tests.cpp)
    target_link_libraries(input_tests PRIVATE portpad_ui)
    target_compile_definitions(input_tests PRIVATE PORTPAD_TEST_KEYMAP="${CMAKE_CURRENT_SOURCE_DIR}/config/keymap.ini")
    add_test(NAME input COMMAND input_tests)
    add_test(NAME startup COMMAND portpad --smoke-test)
    set_tests_properties(input startup PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
